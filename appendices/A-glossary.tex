\chapter{Glossary}

\begin{description}

\item [Angular distance]
    The \emph{angular distance metric} is defined as
    \[
        \Delta R^2 = \Delta\eta^2 + \Delta\varphi^2,
    \]
    where \(\varphi\) is the azimuthal angle around the beam axis.
    %
    This metric approximates the geometric angle between particles in the detector while remaining approximately longitudinally boost invariant.

\item [Cross section]
        For collisions between discrete particles, the \emph{cross section} is defined as the area transverse to their relative motion.
        %
        If the particles were to interact via contact forces (e.g., hard spheres), the cross section corresponds to their geometric size.
        %
        For long range forces however, the cross section is larger than the physical dimensions of the particles due to action--at--a--distance effects.

\item [Differential cross section]
        The \emph{differential cross section} (\(\frac{d\sigma}{d\Omega}\)) provides additional granularity by describing how the probability of scattering depends on specific final state variables, such as scattering angle (\(\theta\)) or energy transfer.
        %
        It is defined as
        \begin{equation}
            \frac{\dd\sigma}{\dd\Omega} = \frac{\text{Number of events scattered into } \dd\Omega}{\text{Incident flux} \times \text{Target density}}.
        \end{equation}

\item [Integrated luminosity]
            The \emph{integrated luminosity} \(\mathcal{L}_{\text{int}}\) is defined as
            \begin{equation}
                \mathcal{L}_{\text{int}}
                = \int_{t_0}^{t_f} \mathcal{L}(t)\,\dd t,
                \qquad
                \sigma
                = \frac{N_{\text{obs}} - N_{\text{bkg}}}
                       {\mathcal{L}_{\text{int}}\,\epsilon\,A}.
              \end{equation}
              Here \(N_{\text{obs}}\) is the number of selected events, \(N_{\text{bkg}}\) an estimate of background contaminations, \(\epsilon\) the detector and selection efficiency and \(A\) the geometric--plus--kinematic acceptance of the analysis.

\item [Luminosity]
            The \emph{luminosity} \(\mathcal{L}(t)\) is defined such that the interaction rate for a process with cross section \(\sigma\) is \(\dd N/\dd t = \mathcal{L}(t)\,\sigma\).

\item [Monte Carlo tuning]
                \emph{Monte Carlo tuning} is the iterative process of adjusting simulation parameters to match measured cross sections, ensuring that detector simulations accurately reproduce real experimental data.

\item [Natural units]
        \emph{Natural Units} are a system of units where \(\hbar = c = 1\), effectively setting the speed of light and quantum of action as fundamental measuring sticks.
        %
        This choice transforms all physical quantities into powers of energy.

\item [Parton distribution functions]
                \emph{Parton distribution functions} (PDFs) are probability distributions that describe the probability of finding a parton (quark or gluon) in a hadron at a given momentum fraction \(x\) and scale \(Q^2\).
                %
                PDFs are determined from global fits to a wide range of hard scattering processes, including deep inelastic scattering, Drell--Yan production, and jet production.

\item [Parton luminosity]
                \emph{Parton luminosity} is the effective luminosity for specific parton--parton interactions, calculated by convolving the total luminosity with parton distribution functions.

\item [Poisson distribution.]
            The \emph{Poisson distribution} is the fundamental distribution governing counting experiments.
            %
            For \(n\) observed events with expected value \(\lambda\),
            \[
                \P(n\mid\lambda) = \frac{\lambda^n e^{-\lambda}}{n!}\;.
            \]

\item [Profile likelihood ratio]
        The \emph{profile likelihood ratio}
            \[
                \mathcal L(\mu) = \frac{L(\mu, \hat\theta_\mu)}{ L(\hat\mu, \hat\theta)}
            \]
            where \(\hat\theta_\mu\) maximises \(L\) for fixed signal strength \(\mu\) is the optimal test statistics for hypothesis testing under the Neyman--Pearson lemma.

\item [Pseudorapidity]
        The \emph{pseudorapidity} \(\eta,\) is defined as
        \[
            \eta = -\ln \tan\frac\theta2
            \]
        where \(\theta\) is the polar angle from the beam axis~\cite{rubbia_phenomenology_2022}.
        %
        Under a longitudinal boost with rapidity \(\beta = \operatorname{artanh}\nicefrac{v}{c}\), a massless particle's \(\eta\) transforms as \(\eta\mapsto\eta+\beta\)

\item [Correlation matrix.]
                A useful, related measure is the \emph{correlation matrix}, \(\vb*{\rho}\), defined as the normalised covariance matrix,
                \begin{equation}
                    \rho_{ij} = \frac{\Sigma_{ij}}{\sqrt{\Sigma_{ii}\Sigma_{jj}}},
                \end{equation}
                where \(\Sigma\) is the correlation matrix.

\item [Covariance matrix.]
                The \emph{covariance matrix} \(\vb*{\Sigma}\) is defined as
                \begin{equation}
                    \Sigma_{ij} := \mathbb{E}[(\hat{t}_i - \mathbb{E}[\hat{t}_i])(\hat{t}_j - \mathbb{E}[\hat{t}_j])].
                \end{equation}

\item [KL divergence]
                \begin{equation}
                    D_{\text{KL}}(p||q) = \int p(x) \log \frac{p(x)}{q(x)} \;\dd x,
                \end{equation}
                where \(p\) is the true distribution and
                \(q\) is the unfolded distribution.

\item [Pull distribution.]
                For a given unfolded bin or parameter \(\theta\), the \emph{pull} is defined as
            \begin{equation}
                \text{Pull}\,{\theta} = \frac{\hat{\theta} - \theta_{\text{true}}}{\sigma_{\hat{\theta}}}
            \end{equation}
            where \(\hat{\theta}\) is the unfolded estimate, \(\theta_{\text{true}}\) is the true value, and \(\sigma_{\hat{\theta}}\) is the reported uncertainty.

\item [Residuals]
                For binned methods, the \emph{bin by bin residual} is defined as
            \begin{equation}
                \delta_i = \hat{t}_i - t_i
            \end{equation}
            where \(\hat{t}_i\) is the unfolded count in bin \(i\) and \(t_i\) is the true count.

\item [VLC Divergence]
                \begin{equation}
                    \Delta_{VLC}(p, q) = \frac{1}{2}\int \frac{(p(\lambda) - q(\lambda))^2}{p(\lambda) + q(\lambda)} \;\dd\lambda.
                \end{equation}

\item [Wasserstein \(\alpha-\)metric]
            \begin{equation}
            W_\alpha(p, q) = \qty(\inf_{\gamma \in \Gamma(p, q)} \iint |x-y|^\alpha d\gamma(x, y))^{\frac1\alpha},
        \end{equation}
        where \(\Gamma(p, q)\) is the set of all joint distributions with marginals \(p\) and \(q\).
        %

\item [Adaptive methods.]
                \emph{Adaptive methods} are optimisation techniques that vary the learning rate \(\eta\) over the course of the training rather than keeping it constant, and allow different learning rates for different dimensions of the input, rather than the traditional scalar learning rate that was constant across dimensions.

\item [Batch]
                A \emph{mini--batch}, often simply called a \emph{batch}, is a subset of the training data used to compute the gradient of the loss function.
                %
                In practice, neural networks are trained using mini--batch gradient descent, where parameter updates are performed using gradients computed on subsets (batches) of the training data.

\item [Deep neural network]
            A \emph{deep neural network} is a neural network with depth, \(d > 2\).
            %
            That is to say standard deep feed forward neural network consists of an input layer, \emph{more than one} hidden layers, and an output layer.%

\item [Depth of a neural network]
            The \emph{depth}, \(d = \ell_{\max}\), of a neural network is defined as the depth of its deepest layer.

\item [Depth of a neuron]
            The \emph{depth} of a neuron is defined as the length of the longest path between the input and the neuron.

\item [Divergence]
                The \emph{divergence} \(\operatorname{Div}(f(x;\,W), g(x))\) is a continuous valued proxy for the error between \(f(x;\,W)\) and \(g(x)\).

\item [Empirical risk]
            The \emph{empirical} risk as an approximation of the risk function defined as
            \begin{equation}
            \hat{{R}}(f_\theta) = \frac{1}{N}\sum_{i=1}^{n}L(y_i, f_\theta(x_i))\;,
        \end{equation}
        %
        where \(L\) is the loss function, the empirical expected divergence.

\item [Forward propagation]
                \emph{Forward propagation} is the process of computing the network's output given an input.

\item [Hidden layer]
            Any layer of a neural network except the input layer (often labelled layer \(0\)) and the output layer (\(\ell_{\max}\)) is called a hidden layer.

\item [Layer of a neuron]
            The set of all neurons at depth \(\ell\) is called \emph{layer} \(\ell\) of the network.

\item [Learning rate scheduling]

                \emph{Learning rate scheduling} is the process of dynamically adjusting the learning rate during training.

\item [Loss function]
                The \emph{loss function} \(L\) is defined as the unbiased empirical average divergence between the neural network output and the target output over all training instances.
            \begin{equation}
                L(W) = \frac{1}{|X|}\sum_{i=1}^{|X|}\operatorname{Div}\Big(f(x_i;\,W), g(x_i)\Big).
            \end{equation}

\item [Neuron]
            The fundamental unit of a neural network, a \emph{neuron} or \emph{node}, computes a weighted sum of its inputs followed by a non-linear transformation:
            \begin{equation}
            y = \sigma(w^T x + b),
            \end{equation}
            where \(x \in \mathbb{R}^n\) is the input vector, \(y\in\R^{m}\) is the output vector, \(w \in \mathbb{R}^{n\times m}\) is a weight matrix, \(b \in \mathbb{R}^m\) is a bias term, and \(\sigma(\cdot)\) is a (typically non--linear) activation function.

\item [Physics informed neural network]
    A \emph{physics informed neural network} is a neural network that explicitly incorporates domain knowledge about physical laws, conservation principles, and symmetries.

\item [Risk function.]
            For a space of functions \(f_\theta\) parametrised by \(\theta\) the \emph{risk function} is defined as 
            %
            \begin{equation}
                {R}(f_\theta) = \mathbb{E}_{(x,y) \sim p(X,Y)}[\operatorname{Div}(y, f_\theta(x))]\;,
            \end{equation}
            %
            where \(\operatorname{Div}\) is the divergence between the predicted output \(f_\theta(x)\) and the true output \(y\) y, and \(P(X,Y)\) is the joint probability distribution of inputs and outputs.

\item [Spectral normalisation]
        \emph{Spectral normalisation} is the process of controlling the Lipschitz constant of a discriminator by normalising its weights,
        \begin{equation}
            W_{\text{SN}} = \frac{W}{\sigma(W)},
        \end{equation}
        where \(\sigma(W)\) is the spectral norm of weight matrix \(W\), provides additional tools to avoid training instabilities in GANs~\cite{Terjek2019AdversarialRegularization, miyato_spectral_2018}.

\item [Stochastic gradient descent (SGD)]
            
            \emph{Stochastic gradient descent} is an adaptation of gradient descent that updates parameters using gradients computed on mini--batches, introducing noise that can help escape local minima.
            
            \begin{equation}
            W_{t+1} = W_t - \eta\, \grad L_{\text{batch}}(W_t)\;.
            \end{equation}

\item [Width of a layer]
            The \emph{width} of a layer is the number of neurons in that layer.

\item [Binned unfolding]
    A \emph{binned unfolding method} is one in which both the particle level and detector level distributions are discretised into histograms, transforming the continuous convolution of \cref{eq:forward-folding} into the matrix equation \cref{eq:forward-binned}.

\item [Event shape observables.]
                \emph{Event shape observables,} are observables which characterise the geometrical properties of energy flow in collisions, are often studied through their moments.

\item [MLC loss]
 Consider the asymptotic \emph{maximum likelihood classifier} (MLC) loss
\begin{equation}
    L(d,g) = -\int p(z)\,\ln d(z)\,\dd z - \int \qt(z)\big(1-d(z)\big)\,\dd z,
\end{equation}

where $d:\mathbb{R}\to(0,1]$ is a discriminator.

\item [Moments of PDFs.]
                For a parton distribution function $f(x,Q^2)$, where $x$ is the momentum fraction and $Q^2$ is the energy scale, the \(k^{\text{th}}\) moment is defined as
            \[
                M_k(Q^2) = \int_0^1 x^{n-1}\;f(k,Q^2) \dd x.
            \]
            
            The DGLAP equations for these moments take the form
            \[
            \frac{\dd M_k(Q^2)}{\dd \ln Q^2} = \frac{\alpha_s(Q^2)}{2\pi}\; P_k\;M_k(Q^2).
            \]
            where $P_k$ are the moments of the splitting functions~\cite{Ellis:1996mzs, salam_elements_2011, Schoeffel:1998tz, Strozik-Kotlorz:2014xga, Simonelli:2024vyh}.
            %
            This formulation converts the complex integro--differential evolution equations into a set of simple ordinary differential equations, one for each moment independently.

\item [Statistical moment.]
        The \emph{$k^{\text{th}}$ moment} of random variable \(Z\) a probability density $p(z)$ is defined as
        \[
            \E [Z^k] = \int_{-\infty}^{\infty} z^k\, p(z)\, \dd z\;.
        \]

\item 
            [Moment generating function (MGF)]
            The\emph{ moment generating function (MGF)} of a random variable \(Z, M_Z(t) = \mathbb{E}[e^{tZ}],\) is defined as
            \[
                M_Z(t) \;=\; \sum_{n=0}^\infty \frac{t^n}{n!}\,\mathbb{E}[Z^n]\;.
            \]
            The MGF encapsulates the entire moment sequence via its Taylor expansion, and serves as a proxy for the full distribution.

\item [Effective sample size]
    The \emph{effective size} of a sample is the effective number of independent points, $N_{\text{eff}} = N/[1+(N-1)\rho]$~\cite{thompson_power_2021, wang_power_2022, baumann_tutorial_2025, Yang2011EffectiveAnalyses} in that sample accounting for correlations.

\item [Event independence in HEP]
    In high energy physics cross section measurements, standard statistical treatments for unbinned inference rely on a set of core assumptions collectively referred to as \emph{event independence}. 
    %
    These assumptions posit that interaction events are generated and observed as independent trials of a stochastic process, which greatly simplifies the construction of likelihood functions and inference procedures.

\item [Pileup]
        At particle physics experiments multiple events can occur in the same same detector read-out window and be recorded together.
        %
        This phenomenon is known as \emph{pileup}.
        %
        Pileup entails that what is recorded as a single `event' may in fact be a composite of several independent interactions superimposed in the detector readout.

\item [Poisson bootstrap]
                The \emph{Poisson bootstrap} is a variant of non-parametric bootstrap resampling in which instead of drawing exactly $N$ samples each time, each original event, \(x_i\), is assigned a Poisson(1) random weight, \(w_{ij}\) in each replica, \(j\)~\cite{Lyons:2011cli, Bohm:2013gla}.
                %
                \(x_i\) is then included in replica \(j\;w_{ij}\) times.

\item [i.i.d.]
            \emph{Independence and identical distribution} (i.i.d) is the assumption that each event can be treated as an independent draw from the same underlying probability density function $p(x\mid\theta)$, where $x$ denotes the measured observables (kinematic variables, detector signals, etc.) for that event.
            %
            In other words, conditional on the physics model parameters $\theta$, all events are statistically independent and governed by an identical distribution~\cite{Nachman2021EOnce, ParticleDataGroup:2022pth, cowan_statistical_1998, Stakia:2021pvp}.

\item 
            [Symmetry discovery]         \emph{Symmetry discovery} is the process of identifying invariant transformations of a dataset, that is to say, operations under which the probability distribution of the dataset remains unchanged in a statistical sense~\cite{shaw_lie_2025, shaw_symmetry_2024, hagemeyer_learning_2022}.

\item [Diffeomorphism invariant probability densities.]
        Let $X\subseteq\R^n$ be an open set and \(\Phi_G\) be a $C^1$ action of a group $G$ on $X$. 
        %
        Let $\dd x$ denote the Lebesgue measure on $X$.
        %
        A probability density \(p : X \;\to\; [0,\infty)\) is diffeomorphism invariant under the action of \(\Phi_G\) if

\[
    \forall \;\text{measureable} \;A \subseteq X,\; \forall g\in G,\;
    \mu_p\bigl(g\cdot A\bigr)\;=\;\mu_p(A)\;,
\]
where \[
    \mu_p(A)\;=\;\int_A p(x)\,\dd x.
    \]
This definition implies that, for almost every $x\in X$ and for every $g\in G$,
\[
    p\bigl(g(x)\bigr)\,\bigl|\det Dg(x)\bigr| = p(x),
\]
where $Dg(x)$ is the Jacobian of the diffeomorphism $x\mapsto g(x)$.

\item [Equivariant neural network.]
        A neural network \(f_\theta\) is said to be \textit{equivariant} under the action group \(G\) on its input space \(X\) if
        \[
            \forall g\in G\quad\forall x\in X\quad\,f_\theta(g \cdot x) = g \cdot f_\theta(x)\;.
        \]

\item [Group invariant function]
        A function \(f: X \rightarrow \mathbb{R}\) is \(G-\)\emph{invariant} if there exists a function \(\tilde{f}: X/G \rightarrow \R\) with \(f = \tilde{f} \circ \pi\) where \(\pi: X \rightarrow X/G\) is the canonical projection.

\item [Invariant measure]
    For every measure space $(X,\Sigma,\mu)$ and for every group $G$ acting on $X$ through a $\Sigma$-measurable action \(\Phi, \mu\) is called an \emph{invariant measure under \(G\)} if \(\forall (g, A) \in G\times \Sigma\)
    \begin{enumerate}
        \item \(g(A)\in\Sigma\;.\)
        \item     \(
    \mu(g(A))=\mu(A)
    \)
    \end{enumerate}
    Equivalently, using the pushforward \(g_\#(\mu(A))\coloneqq \mu(g\inv(A)), \mu\) is defined to be invariant under \(G\) if
    \[
    \forall (A, g) \in\Sigma\times G\; g_\#(\mu(A)) = \mu(A).
    \]

\item [Reference density]
        A \emph{reference density \(\rho(x)\)} is a probability density function that establishes a canonical measure for comparing probability distributions by enabling the definition of relative entropy
        \[
            H_\rho[X] = -\mathbb{E}_\mu\qty[\ln\frac{\dd\mu}{\dd\rho}].
        \]

\item [Relative symmetry.]
        A diffeomorphism $g$ is defined to be a symmetry of a density $p$ \emph{relative} to $p_I$ if both $p$ and $p_I$ are preserved.
    \begin{equation}
        p\bigl(g(x)\bigr)\,\bigl|\det Dg(x)\bigr|
        \;=\; p(x),
        \quad
        p_I\bigl(g(x)\bigr)\,\bigl|\det Dg(x)\bigr|
        \;=\; p_I(x).
    \end{equation}

\item [Symmetric action of a group~\cite{luzzatto_chapter_2006}.]
Let $(\Omega,\mathcal{F},\mu)$ be a probability space and $G$ a group.  \(G\) is defined to have a \emph{symmetric action} on the measure space $(\Omega,\mathcal{F})$ if there exists a function
\[
    \phi: G \times \Omega \to\Omega,\quad (g,\omega)\mapsto g\cdot\omega
\]
such that
\begin{enumerate}
  \item \[\forall\omega \in\Omega\quad\phi(1, \omega) = \omega\]
  \item \[\forall\omega\in\Omega\quad\forall g_1, g_2\in G\quad \phi(g_1g_2, \omega) = \phi(g_1 (\phi (g_2, \omega))\]
  \item 
\[
    \forall g\in G\quad\forall\,A\in\mathcal{F}\quad\mu\bigl(g\cdot A\bigr)=\mu(A),
\]
\end{enumerate}

\item [Symmetry of a function]
        In group theoretic terms, a \emph{function \(f: X\to Y\) is symmetric} under a group \(G\) there exists a group action of \(G\) on \(X\) 
        \[
        \begin{split}
            \Phi: G\times X\to X\\
                (g, x) \mapsto g(x)
        \end{split}\]
       such that for every $x \in X$ and every $g \in G$, the value of the function \(f(x) = f(g(x))\).

\item [Symmetry of a probability density function]
    If $p(x)$ denotes a probability density on $X$, a group $G$ is a \emph{symmetry of \(p\)} if
    \[
        \forall g \in G \; p(x) \; \dd x = p(g(x)) \; \dd(g(x)).
    \]

\item [Symmetry under a group.]
    Let $G$ be a group of transformations (continuous or discrete) acting on a space of states or observations $X$.
    %
    A physical system or probability distribution is \emph{symmetric under $G$} if applying any transformation $g \in G$ leaves the relevant observables unchanged.

\item [{\(G\)-Equivariant Fisher Information}]
                For a likelihood \(\mathcal L(x\mid\theta)\) whose parameter space carries an action of a finite or Lie group \(G\), the \emph{\(G\)-equivariant Fisher information matrix} is  
                \[
                  \bigl[\mathcal I_G(\theta)\bigr]_{ij}
                  \;=\;
                  \mathbb E\!\Bigl[
                      \frac{\partial\log \mathcal L}{\partial\theta_i}
                      \frac{\partial\log \mathcal L}{\partial\theta_j}
                  \Bigr]
                  \;-\;
                  \sum_{g\in G} w_g\,
                  \mathbb E\!\Bigl[
                      \frac{\partial\log \mathcal L}{\partial\theta_i}
                      \frac{\partial\log \mathcal L(g \cdot \theta)}{\partial\theta_j}
                  \Bigr],
                \]
                where \(\{w_g\}_{g\in G}\) are averaging weights satisfying \(\sum_{g\in G}w_g=1\).  Equivalently,  
                \(\mathcal I_G(\theta)=P_G\,\mathcal I(\theta)\,P_G^{T}\) where \(P_G\) is the orthogonal projection operator onto the \(G\)-invariant subspace~\cite{nielsen_elementary_2020}.
\end{description}
